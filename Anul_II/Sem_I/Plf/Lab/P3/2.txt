descompunere(l1 l2 ... lm, s) =
	1. (l1)						, daca l1 = s
	2. descompunere(l2 ... lm, s)
	3. l1 (+) descompunere(l2 ... lm, s - l1)	, daca s - l1 > 0


descompunerePrime(n) = descompunere(listaPrime(2, n), n)
